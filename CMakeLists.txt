cmake_minimum_required(VERSION 3.16)
project(AsterVM VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(astervm_core src/machine.cpp src/decode16.cpp src/execute16.cpp src/json.cpp)
target_include_directories(astervm_core PUBLIC include)
add_executable(astervm_cli src/main.cpp)
target_link_libraries(astervm_cli PRIVATE astervm_core)
add_executable(astervm_step src/step_bridge.cpp)
target_link_libraries(astervm_step PRIVATE astervm_core)
if(MSVC)
  target_compile_options(astervm_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(astervm_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(astervm_tests tests/core_tests.cpp)
  target_link_libraries(astervm_tests PRIVATE astervm_core)
  add_test(NAME astervm_core COMMAND astervm_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME astervm_cli_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py" "$<TARGET_FILE:astervm_cli>")
    add_test(NAME astervm_external_m0_profile COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/run_external_qa.py" --bridge "$<TARGET_FILE:astervm_step>" --report "${CMAKE_CURRENT_BINARY_DIR}/external-qa-report.json")
  endif()
endif()
